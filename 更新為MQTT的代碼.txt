四、主題發布邏輯
```csharp
// MQTT發布管理器
public class MqttPublisher {
    private IMqttClient _mqttClient;
    
    public async Task PublishDeviceData(DeviceData data) {
        var topic = $"SUNTREE/{DeviceId}/DEVICE/Data";
        var payload = JsonConvert.SerializeObject(data);
        await _mqttClient.PublishAsync(topic, payload);
    }

    // 故障發布示例
    public async Task PublishFault(int lineNumber, int faultType) {
        var faultTopic = $"SUNTREE/{DeviceId}/DEVICE/Fault";
        var faultData = new {
            lineNumber,
            id = Guid.NewGuid().GetHashCode() % 256,
            type = faultType,
            datetime = DateTime.Now.ToString("yyyy/MM/dd,HH:mm:ss"),
            almvalue = GetAlarmValue(faultType) 
        };
        await _mqttClient.PublishAsync(faultTopic, JsonConvert.SerializeObject(faultData));
    }
}
```

五、命令訂閱處理
```csharp
// 命令訂閱管理器
public class MqttCommandHandler {
    public void SubscribeCommands() {
        _mqttClient.SubscribeAsync($"PROJECT/{DeviceId}/CONTROL/#");
    }

    // 處理開關命令
    private void HandleSwitchCommand(string payload) {
        dynamic cmd = JsonConvert.DeserializeObject(payload);
        if (cmd.targetType == 1) {
            ExecuteSingleDeviceSwitch(cmd.lineNumber, cmd.action);
        }
        // 發送ACK
        var ackTopic = $"PROJECT/{DeviceId}/CONTROL/Action/ACK";
        var ackPayload = new {
            id = cmd.id,
            lineNumber = cmd.lineNumber,
            switchState = (cmd.action == 1)  
        };
        _mqttClient.PublishAsync(ackTopic, JsonConvert.SerializeObject(ackPayload));
    }
}
```

六、字段映射關係
| Modbus字段          | MQTT JSON字段       | 轉換邏輯                  |
|---------------------|---------------------|-------------------------|
| currentA           | ia                 | 直接映射，單位轉換(A保留小數點後2位) |
| voltageA           | ua                 | 電壓值直接映射               |
| tempA              | Temp_a             | 溫度字段名稱變化              |
| powerFactorA       | pfa                | 功率因數字段名稱變化            |
| switchStatus       | switchState        | 狀態轉換：1->true, 0->false |
| currentStatus      | type               | 根據錯誤碼映射到對應的故障類型       |
